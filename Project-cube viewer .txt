<body>
  <div id="panel">
    <h1>3D Cube Viewer</h1>
    <label>Speed <input type="range" id="speed" min="0" max="0.05" step="0.001" value="0.01"></label>
    <button id="colorBtn">Change colour</button>
    <button id="wireBtn">Toggle wireframe</button>
    <small>Drag the cube to rotate it</small>
  </div>

  <!-- Three.js library from CDN (needs internet) -->
  <script src="https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js"></script>
  <script>
    // 1. Scene, camera, renderer
    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 100);
    camera.position.z = 4;

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(window.innerWidth, window.innerHeight);
    document.body.appendChild(renderer.domElement);

    // 2. Cube (geometry + material = mesh)
    const material = new THREE.MeshStandardMaterial({ color: 0xfca311 });
    const cube = new THREE.Mesh(new THREE.BoxGeometry(1.5, 1.5, 1.5), material);
    scene.add(cube);

    // 3. Lights
    scene.add(new THREE.AmbientLight(0xffffff, 0.5));
    const light = new THREE.DirectionalLight(0xffffff, 0.9);
    light.position.set(3, 4, 5);
    scene.add(light);

    // 4. Controls
    let speed = 0.01;
    document.getElementById("speed").addEventListener("input", e => speed = parseFloat(e.target.value));
    document.getElementById("colorBtn").addEventListener("click", () => material.color.setHex(Math.random() * 0xffffff));
    document.getElementById("wireBtn").addEventListener("click", () => material.wireframe = !material.wireframe);

    // 5. Drag to rotate
    let dragging = false, lastX = 0, lastY = 0;
    const canvas = renderer.domElement;
    canvas.addEventListener("pointerdown", e => { dragging = true; lastX = e.clientX; lastY = e.clientY; });
    window.addEventListener("pointerup", () => dragging = false);
    window.addEventListener("pointermove", e => {
      if (!dragging) return;
      cube.rotation.y += (e.clientX - lastX) * 0.01;
      cube.rotation.x += (e.clientY - lastY) * 0.01;
      lastX = e.clientX; lastY = e.clientY;
    });

    // 6. Resize handling
    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    // 7. Animation loop
    function animate() {
      requestAnimationFrame(animate);
      if (!dragging) { cube.rotation.x += speed; cube.rotation.y += speed; }
      renderer.render(scene, camera);
    }
    animate();
  </script>
